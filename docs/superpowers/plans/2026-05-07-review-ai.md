In `crates/opt/src/known_bits/mod.rs`, after the existing `NodeKind::Extend(ExtendOp::ZeroExtend)` arm, add:

```rust
NodeKind::Extend(ExtendOp::SignExtend) => {
    let [input] = fg.graph.node_inputs_exact::<1>(node_id)?;
    let input_kind = fg.graph.output_kind(input);
    let input_ty = input_kind.as_value_or_err()?;
    let Some(input_mask) = u64_type_mask(input_ty) else { return Ok(None) };
    let kb = known.get(&input).copied().unwrap_or_default();
    let sign_bit = (input_mask >> 1) + 1;
    let upper_mask = type_mask & !input_mask;
    if kb.ones & sign_bit != 0 {
        Kb { ones: kb.ones | upper_mask, zeros: kb.zeros }
    } else if kb.zeros & sign_bit != 0 {
        Kb { ones: kb.ones, zeros: kb.zeros | upper_mask }
    } else {
        Kb { ones: kb.ones & input_mask, zeros: kb.zeros & input_mask }
    }
}
```
